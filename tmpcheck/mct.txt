function chart() {
  const cv = $('#ch'), c = COIN.coin, dpr = window.devicePixelRatio || 1, W = cv.clientWidth, H = cv.clientHeight;
  cv.width = W * dpr; cv.height = H * dpr; const g = cv.getContext('2d'); g.scale(dpr, dpr);
  let pts = (c.real ? [] : [[Date.parse(c.created_at), P0]]).concat(COIN.trades.map((x) => [x[0], x[1]])); pts.push([Date.now(), c.price]);
  if (pts.length < 2) pts.unshift([Date.now() - 36e5, c.price]);
  const t0 = pts[0][0], t1 = Math.max(pts[pts.length - 1][0], t0 + 1), lo = Math.min(...pts.map((p) => p[1])), hi = Math.max(...pts.map((p) => p[1])), pad = (hi - lo) * 0.15 || hi * 0.1;
  const X = (x) => 16 + (W - 32) * (x - t0) / (t1 - t0), Y = (v) => 12 + (H - 24) * (1 - (v - lo + pad) / (hi - lo + 2 * pad));
  const col = c.color || '#ffd43b';
  g.beginPath(); pts.forEach((p, i) => i ? g.lineTo(X(p[0]), Y(p[1])) : g.moveTo(X(p[0]), Y(p[1])));
  const line = new Path2D(); pts.forEach((p, i) => i ? line.lineTo(X(p[0]), Y(p[1])) : line.moveTo(X(p[0]), Y(p[1])));
  g.lineTo(X(t1), H); g.lineTo(X(t0), H); g.closePath();
  const gr = g.createLinearGradient(0, 0, 0, H); gr.addColorStop(0, col + '66'); gr.addColorStop(1, col + '00'); g.fillStyle = gr; g.fill();
  g.lineWidth = 3; g.lineJoin = 'round'; g.strokeStyle = col; g.shadowColor = col; g.shadowBlur = 12; g.stroke(line); g.shadowBlur = 0;
  const lp = pts[pts.length - 1]; g.beginPath(); g.arc(X(lp[0]), Y(lp[1]), 5, 0, 7); g.fillStyle = '#fff'; g.fill();
}
function burst(from, emoji) {
  if (matchMedia('(prefers-reduced-motion: reduce)').matches) return;
  const r = from.getBoundingClientRect();
  for (let i = 0; i < 14; i++) { const s = document.createElement('span'); s.className = 'burst'; s.textContent = i % 3 ? emoji : '💵'; s.style.left = (r.left + r.width / 2 - 12) + 'px'; s.style.top = (r.top + r.height / 2 - 12) + 'px'; const a = Math.random() * Math.PI * 2, d = 80 + Math.random() * 120; s.style.setProperty('--dx', Math.cos(a) * d + 'px'); s.style.setProperty('--dy', (Math.sin(a) * d - 60) + 'px'); s.style.setProperty('--r', (Math.random() * 360 - 180) + 'deg'); document.body.appendChild(s); setTimeout(() => s.remove(), 1000); }
}
